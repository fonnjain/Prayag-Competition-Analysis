0.2 — One normalization rule

The audit imports normalizeVocabularyKey from the existing P048 module. The actual rule includes NFKC Unicode normalization and singular/plural ending reduction in addition to trim, whitespace collapse, case folding and punctuation/symbol stripping. Thus Cygnus normalizes to “cygnu”, Marcus to “marcu”, and Fittings to “fitting”. The rule has not been changed or copied.

export const normalizeVocabularyKey = (value: string) => value.normalize("NFKC").trim().replace(/\s+/g, " ")
  .toLocaleLowerCase("en").replace(/[\p{P}\p{S}]/gu, " ").trim().replace(/\s+/g, " ")
  .split(" ").filter(Boolean).map((word) => {
    if (word.length > 3 && word.endsWith("ies")) return `${word.slice(0, -3)}y`;
    if (word.length > 3 && word.endsWith("s") && !word.endsWith("ss")) return word.slice(0, -1);
    return word;
  }).join(" ");
